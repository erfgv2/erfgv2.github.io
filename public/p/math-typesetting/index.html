import math
from Crypto.Util.number import long_to_bytes
a = math.isqrt(n) + 1
b2 = a * a - n
b = math.isqrt(b2)
while b * b != b2:
    a += 1
    b2 = a * a - n
    b = math.isqrt(b2)
p, q = a - b, a + b
print(p)
print(q)
p1 = (p - 1) * (q - 1)
d = pow(e, -1, p1)
m = pow(c, d, n)
print(m)
print(long_to_bytes(m))
